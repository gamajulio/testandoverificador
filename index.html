async function geocodificar(enderecoOriginal) { // 1. Limpeza e correção automática de erros comuns de digitação/abreviações let enderecoTratado = enderecoOriginal .replace(/\b(av|av\.)\b/gi, "Avenida") .replace(/\b(r|r\.)\b/gi, "Rua") .replace(/\b(tr|trav\.)\b/gi, "Travessa") .replace(/\b(rod\.)\b/gi, "Rodovia") .trim(); // Se o usuário não colocou a cidade, força a busca na Paraíba para ajudar o mapa a achar const temLocalidade = /joão pessoa|bayeux|santa rita|cabedelo|pb/i.test(enderecoTratado); const queryCompleta = temLocalidade ? enderecoTratado : enderecoTratado + ", Paraíba"; // Parâmetro 'amenity' ou 'addressdetails' ajudam, mas o segredo para erros de digitação // é usar a busca padrão do Nominatim que já possui tolerância ortográfica nativa (fuzzy search), // complementada por uma busca de fallback se a primeira falhar. const url = `https://nominatim.openstreetmap.org/search?format=json&addressdetails=1&q=` + encodeURIComponent(queryCompleta); try { let resposta = await fetch(url, { headers: { 'User-Agent': 'CalculadoraFreteZeny/8.0' } }); let dados = await resposta.json(); // SE NÃO ACHOX DE PRIMEIRA (geralmente por erro de digitação grave), // tentamos uma segunda busca mais ampla apenas com as principais palavras da rua + bairro/cidade if (!dados || dados.length === 0) { // Tenta remover números e buscar de forma mais ampla se falhou na primeira const palavrasChave = enderecoTratado.replace(/[0-9]/g, '').trim(); const urlAmpla = `https://nominatim.openstreetmap.org/search?format=json&addressdetails=1&q=` + encodeURIComponent(palavrasChave + ", Paraíba"); const respostaAmpla = await fetch(urlAmpla, { headers: { 'User-Agent': 'CalculadoraFreteZeny/8.0' } }); dados = await respostaAmpla.json(); } if (dados && dados.length > 0) { const cidadesPermitidas = ['joão pessoa', 'bayeux', 'santa rita', 'cabedelo']; for (let item of dados) { const addr = item.address || {}; const cidade = (addr.city || addr.town || addr.municipality || addr.city_district || '').toLowerCase(); if (cidadesPermitidas.some(cidadeValida => cidade.includes(cidadeValida))) { return { lat: parseFloat(item.lat), lon: parseFloat(item.lon) }; } } return null; } return null; } catch (e) { console.error("Erro na geocodificação com tolerância:", e); return null; } }
